            // 根据多个答案组合选择文案组 (使用 q1,q4,q7,q9,q12)
            function selectTraitSet(answers) {
                const q1 = answers.q1 || '';
                const q4 = answers.q4 || '';
                const q7 = answers.q7 || '';
                const q9 = answers.q9 || '';
                const q12 = answers.q12 || '';

                let score = 0;
                // q1: 都市/海边 → 浪漫型(2) 或 社交型(9); 小镇/山林 → 深邃型(4) 或 哲学型(8)
                if (q1.includes('都市')) score += 2;
                else if (q1.includes('海边')) score += 3;
                else if (q1.includes('小镇')) score += 1;
                else if (q1.includes('山林')) score += 4;

                // q4: 直觉 → 敏锐(0) 或 浪漫(2); 逻辑 → 理性(1) 或 清醒(3); 结合 → 创意(5) 或 哲学(8)
                if (q4.includes('直觉')) score += 1;
                else if (q4.includes('逻辑')) score += 2;
                else if (q4.includes('结合')) score += 3;

                // q7: 善良 → 温柔(6); 聪明 → 理性(1) 或 创意(5); 勇敢 → 冒险(7); 幽默 → 社交(9)
                if (q7.includes('善良')) score += 2;
                else if (q7.includes('聪明')) score += 1;
                else if (q7.includes('勇敢')) score += 4;
                else if (q7.includes('幽默')) score += 5;

                // q9: 过去 → 哲学(8) 或 浪漫(2); 现在 → 冒险(7) 或 温柔(6); 未来 → 创意(5) 或 理性(1)
                if (q9.includes('过去')) score += 1;
                else if (q9.includes('现在')) score += 2;
                else if (q9.includes('未来')) score += 3;

                // q12: 爱情 → 浪漫(2) 或 温柔(6); 事业 → 理性(1) 或 冒险(7); 平衡 → 社交(9) 或 创意(5)
                if (q12.includes('爱情')) score += 2;
                else if (q12.includes('事业')) score += 1;
                else if (q12.includes('平衡')) score += 3;

                const idx = (score % 10 + 10) % 10;
                return TRAIT_SETS[idx];
            }

                if (!birthDate) {
                    alert('请填写出生日期 (必填)');
                    return;
                }
                const answers = getAnswers();
                let filled = 0;
                for (let key in answers) {
                    if (answers[key] && answers[key].length > 0) filled++;
                }
                if (filled < 3) {
                    alert('请至少回答几个问题，让星盘更准确 ✦');
                    return;
                }

                const birthTime = document.getElementById('birthTime').value;
                const birthPlace = document.getElementById('birthPlace').value;
                const report = generateReport(birthDate, birthTime, birthPlace, answers);
                displayResult(report);
            });
